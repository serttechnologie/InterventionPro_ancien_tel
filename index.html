# ================================ # Nettoyage automatique Chrome # Windows 10 # ================================ $logFile = "$env:USERPROFILE\Documents\nettoyage_chrome_log.txt" function Log($message) { Add-Content $logFile "[$(Get-Date -Format 'dd/MM/yyyy HH:mm:ss')] $message" } Log "----- DÉBUT DU SCRIPT -----" try { # Fermer Chrome s'il est ouvert Log "Fermeture de Chrome" Get-Process chrome -ErrorAction SilentlyContinue | Stop-Process -Force Start-Sleep -Seconds 2 # Chemin Chrome $chromePath = "$env:LOCALAPPDATA\Google\Chrome\User Data\Default" # Suppression cookies et cache Log "Suppression cookies et cache Chrome" Remove-Item "$chromePath\Cookies" -Force -ErrorAction SilentlyContinue Remove-Item "$chromePath\Cache\*" -Recurse -Force -ErrorAction SilentlyContinue Remove-Item "$chromePath\Code Cache\*" -Recurse -Force -ErrorAction SilentlyContinue Remove-Item "$chromePath\GPUCache\*" -Recurse -Force -ErrorAction SilentlyContinue # Nettoyage fichiers temporaires Windows Log "Nettoyage fichiers temporaires Windows" Remove-Item "$env:TEMP\*" -Recurse -Force -ErrorAction SilentlyContinue Remove-Item "C:\Windows\Temp\*" -Recurse -Force -ErrorAction SilentlyContinue # Relancer Chrome Log "Lancement de Chrome" Start-Process "chrome.exe" Log "SCRIPT TERMINÉ AVEC SUCCÈS ✅" } catch { Log "ERREUR ❌ : $_" } Log "----- FIN DU SCRIPT -----`n"
